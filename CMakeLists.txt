cmake_minimum_required(VERSION 3.24)
project(AlienStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(AlienSynthTests tests/SynthTests.cpp)
    target_include_directories(AlienSynthTests PRIVATE src)
    target_link_libraries(AlienSynthTests PRIVATE Threads::Threads)
    add_test(NAME AlienSynthTests COMMAND AlienSynthTests)
    set_tests_properties(AlienSynthTests PROPERTIES TIMEOUT 30)
endif()

option(ALIEN_BUILD_APP "Build the JUCE desktop application" ON)
if(NOT ALIEN_BUILD_APP)
    return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(AlienStudio PRODUCT_NAME "Alien Studio" COMPANY_NAME "Alien Studio")
target_sources(AlienStudio PRIVATE src/Main.cpp src/synth/SynthEngine.h
    src/audio/AudioEngine.h src/audio/SequencerEngine.h src/audio/ControlQueue.h)
target_compile_definitions(AlienStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(AlienStudio PRIVATE juce::juce_audio_utils juce::juce_gui_extra)
juce_generate_juce_header(AlienStudio)
