cmake_minimum_required(VERSION 3.24)
project(AlienStudio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(AlienStudio PRODUCT_NAME "Alien Studio" COMPANY_NAME "Alien Studio" BUNDLE_ID "org.alienstudio.daw")
target_sources(AlienStudio PRIVATE src/Main.cpp)
target_compile_definitions(AlienStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
# Branding artwork embedded for the splash screen and in-app logo (the Windows
# executable icon is embedded separately from the ICO below).
juce_add_binary_data(AlienStudioBranding HEADER_NAME BrandingData.h NAMESPACE BrandingData
    SOURCES assets/branding/AlienStudio_Icon_Corrected.png)
target_link_libraries(AlienStudio PRIVATE AlienStudioBranding juce::juce_audio_processors juce::juce_audio_utils juce::juce_gui_extra)
juce_generate_juce_header(AlienStudio)

if(WIN32)
    enable_language(RC)
    set(ALIEN_STUDIO_WINDOWS_ICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/AlienStudio_Icon_Corrected.ico")
    configure_file(resources/AlienStudioIcons.rc.in
        "${CMAKE_CURRENT_BINARY_DIR}/AlienStudioIcons.rc" @ONLY)
    # Keep JUCE's generated VERSIONINFO, and compile the supplied ICO unchanged.
    target_sources(AlienStudio PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/AlienStudioIcons.rc")
    set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/AlienStudioIcons.rc" APPEND
        PROPERTY OBJECT_DEPENDS "${ALIEN_STUDIO_WINDOWS_ICON}")
endif()
